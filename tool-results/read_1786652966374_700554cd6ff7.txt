    35→/** Map a 0..1 dimension score onto the smooth color scale:
    36→ * <0.30 critical · 0.30–0.50 warning · 0.50–0.70 accent-blue · >0.70 success. */
    37→function scoreToStatus(score: number): "matched" | "weak" | "gap" | "accent" {
    38→  if (score >= 0.7) return "matched";
    39→  if (score >= 0.5) return "accent";
    40→  if (score >= 0.3) return "weak";
    41→  return "gap";
    42→}

   495→function InsightsTabs() {
   496→  const { gaps, interview, match } = useHireMind();
   497→  const [tab, setTab] = React.useState("strengths");
   498→
   499→  // Derive strengths: competencies whose current level is strong or moderate,
   500→  // or that have interview evidence with high overall scores.
   501→  const strengths: { name: string; reason: string }[] = [];
   502→  const watchOuts: { name: string; reason: string }[] = [];
   503→
   504→  if (interview) {
   505→    for (const c of interview.competencyStates) {
   506→      if (c.current === "strong") {
   507→        strengths.push({
   508→          name: c.competency,
   509→          reason:
   510→            c.interviewLevel !== "unknown"
   511→              ? `Demonstrated in interview (resume: ${c.resumeLevel}).`
   512→              : `Strong resume signal — not yet verified in interview.`,
   513→        });
   514→      } else if (c.current === "moderate" && c.interviewLevel !== "unknown") {
   515→        strengths.push({
   516→          name: c.competency,
   517→          reason: `Solid interview evidence (resume: ${c.resumeLevel} → interview: ${c.interviewLevel}).`,
   518→        });
   519→      } else if (c.status === "gap" || c.current === "unknown" || c.current === "weak") {
   520→        watchOuts.push({
   521→          name: c.competency,
   522→          reason:
   523→            c.notes ||
   524→            (c.interviewLevel === "unknown"
   525→              ? "No interview evidence yet — unverified."
   526→              : "Low scores in this area."),
   527→        });
   528→      }
   529→    }
   530→  }
   531→
   532→  // Augment watch-outs from gaps list (if not already present)
   533→  if (gaps) {
   534→    for (const g of gaps) {
   535→      if (watchOuts.find((w) => w.name === g.competency)) continue;
   536→      if (g.importance === "critical" || g.importance === "high") {
   537→        watchOuts.push({
   538→          name: g.competency,
   539→          reason: g.reason,
   540→        });
   541→      }
   542→    }
   543→  }
   544→
   545→  // Augment strengths from match rows (matched competencies not in competencyStates)
   546→  if (match) {
   547→    for (const r of match.rows) {
   548→      if (r.status === "matched" && !strengths.find((s) => s.name === r.competency)) {
   549→        strengths.push({
   550→          name: r.competency,
   551→          reason: r.evidence
   552→            ? `Resume evidence: "${r.evidence.slice(0, 80)}${r.evidence.length > 80 ? "…" : ""}"`
   553→            : "Aligned with job requirements.",
   554→        });
   555→      }
   556→    }
   557→  }
   558→
   559→  // Cap each list at 6 items for layout stability
   560→  const topStrengths = strengths.slice(0, 6);
   561→  const topWatchOuts = watchOuts.slice(0, 6);
   562→
   563→  return (
   564→    <motion.div
   565→      initial={{ opacity: 0, y: 8 }}
   566→      animate={{ opacity: 1, y: 0 }}
   567→      transition={{ duration: 0.5, delay: 0.18 }}
   568→      className="hm-card hm-card-hover mt-4 p-4 sm:p-6 overflow-visible"
   569→    >
   570→      <div className="flex items-center gap-2.5 mb-4">
   571→        <span className="inline-flex h-9 w-9 items-center justify-center rounded-xl bg-accent-blue/15 text-accent-blue-foreground">
   572→          <LayoutGrid className="h-5 w-5" />
   573→        </span>
   574→        <div>
   575→          <h3 className="text-[13px] font-semibold">Insights</h3>
   576→          <div className="text-[11px] text-muted-foreground">A deeper breakdown of your readiness signals</div>
   577→        </div>
   578→      </div>
   579→
   580→      <Tabs value={tab} onValueChange={setTab}>
   581→        <TabsList className="mb-4">
   582→          <TabsTrigger value="strengths" className="gap-1.5">
   583→            <CheckCircle2 className="h-3 w-3" />
   584→            Strengths
   585→            <span className="ml-1 rounded bg-success/15 text-success-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   586→              {topStrengths.length}
   587→            </span>
   588→          </TabsTrigger>
   589→          <TabsTrigger value="watchouts" className="gap-1.5">
   590→            <AlertTriangle className="h-3 w-3" />
   591→            Watch-outs
   592→            <span className="ml-1 rounded bg-warning/15 text-warning-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   593→              {topWatchOuts.length}
   594→            </span>
   595→          </TabsTrigger>
   596→          <TabsTrigger value="coverage" className="gap-1.5">
   597→            <LayoutGrid className="h-3 w-3" />
   598→            Coverage
   599→          </TabsTrigger>
   600→        </TabsList>
   601→
   602→        {/* Strengths tab */}
   603→        <TabsContent value="strengths">
   604→          {topStrengths.length > 0 ? (
   605→            <ul className="grid sm:grid-cols-2 gap-2.5">
   606→              {topStrengths.map((s, i) => (
   607→                <motion.li
   608→                  key={s.name}
   609→                  initial={{ opacity: 0, y: 6 }}
   610→                  animate={{ opacity: 1, y: 0 }}
   611→                  transition={{ duration: 0.3, delay: i * 0.05 }}
   612→                  className="flex items-start gap-2.5 rounded-xl border border-success/25 bg-success/5 p-3"
   613→                >
   614→                  <span className="mt-0.5 inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-md bg-success/15 text-success-foreground">
   615→                    <CheckCircle2 className="h-3 w-3" />
   616→                  </span>
   617→                  <div className="min-w-0">
   618→                    <div className="text-[13px] font-semibold text-foreground leading-tight">{s.name}</div>
   619→                    <div className="mt-0.5 text-[11px] text-muted-foreground leading-relaxed">{s.reason}</div>
   620→                  </div>
   621→                </motion.li>
   622→              ))}
   623→            </ul>
   624→          ) : (
   625→            <p className="text-[13px] text-muted-foreground py-6 text-center">
   626→              No standout strengths detected yet. Run the interview to gather evidence.
   627→            </p>
   628→          )}
   629→        </TabsContent>
   630→
   631→        {/* Watch-outs tab */}
   632→        <TabsContent value="watchouts">
   633→          {topWatchOuts.length > 0 ? (
   634→            <ul className="grid sm:grid-cols-2 gap-2.5">

   636→                <motion.li

   639→                  animate={{ opacity: 1, y: 0 }}
   640→                  transition={{ duration: 0.3, delay: i * 0.05 }}

   669→/** Coverage heatmap — grid of competency cells. Each cell shows the competency
   670→ *  name, level, and required badge. Background tint + dot color reflect the
   671→ *  candidate's current level. */
   672→function CoverageHeatmap() {
   673→  const { match, interview } = useHireMind();
   674→
   675→  // Merge match rows with interview competency states so each cell shows the
   676→  // latest known level (interview evidence takes precedence over resume level).
   677→  const interviewLevels = new Map<string, SkillLevel>();
   678→  if (interview) {
   679→    for (const c of interview.competencyStates) {
   680→      interviewLevels.set(c.competency, c.current);
   681→    }
   682→  }
   683→
   684→  if (!match || match.rows.length === 0) {
   685→    return (
   686→      <p className="text-[13px] text-muted-foreground py-6 text-center">
   687→        Run an analysis to see your competency coverage.
   688→      </p>
   689→    );
   690→  }
   691→
   692→  const rows = match.rows;
   693→
   694→  return (
   695→    <div>
   696→      <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   697→        <div className="text-[11px] text-muted-foreground">
   698→          {rows.length} competencies · color shows your level · ring shows required
   699→        </div>
   700→        <div className="flex items-center gap-3 text-[10px] text-muted-foreground">

   702→            <span key={lvl} className="inline-flex items-center gap-1">

   707→        </div>
