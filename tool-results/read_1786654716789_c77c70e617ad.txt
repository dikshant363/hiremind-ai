    46→    46→
    47→    47→function average(nums: number[]): number {
    48→    48→  if (nums.length === 0) return 0;
    49→    49→  return nums.reduce((a, b) => a + b, 0) / nums.length;
    50→    50→}

    52→    52→/** Build radar dimension values from the cumulative evaluations. Each value
    53→    53→ *  is 0-100. */
    54→    54→function buildRadarValues(evals: AnswerEvaluation[]): Record<RadarKey, number> {
    55→    55→  const v: Record<RadarKey, number> = {
    56→    56→    technicalAccuracy: average(evals.map((e) => e.technicalAccuracy)) * 100,
    57→    57→    relevance: average(evals.map((e) => e.relevance)) * 100,
    58→    58→    depth: average(evals.map((e) => e.depth)) * 100,
    59→    59→    communication: average(evals.map((e) => e.communication)) * 100,
    60→    60→    problemSolving:
    61→    61→      average(
    62→    62→        evals.map((e) => (e.technicalAccuracy * 0.5 + e.depth * 0.5))
    63→    63→      ) * 100,
    64→    64→  };
    65→    65→  return v;
    66→    66→}

    68→    68→/** Estimate minutes spent per answer from character count assuming ~30 wpm
    69→    69→ *  and ~5 chars/word -> ~150 chars/min. */
    70→    70→function estimateMinutes(text: string): number {
    71→    71→  const chars = text.trim().length;
    72→    72→  if (chars === 0) return 0;
    73→    73→  return Math.max(0.3, Math.round((chars / 150) * 10) / 10);
    74→    74→}

    76→    76→/** Build a smooth catmull-rom spline path string from a list of (x, y)
    77→    77→ *  points. Tension defaults to 0.5 for a gentle curve. */
    78→    78→function smoothPath(
    79→    79→  points: { x: number; y: number }[],
    80→    80→  tension = 0.5
    81→    81→): string {
    82→    82→  if (points.length === 0) return "";
    83→    83→  if (points.length === 1) return `M ${points[0].x} ${points[0].y}`;
    84→    84→  if (points.length === 2) return `M ${points[0].x} ${points[0].y} L ${points[1].x} ${points[1].y}`;
    85→    85→
    86→    86→  let d = `M ${points[0].x} ${points[0].y}`;
    87→    87→  for (let i = 0; i < points.length - 1; i++) {
    88→    88→    const p0 = points[i - 1] ?? points[i];
    89→    89→    const p1 = points[i];
    90→    90→    const p2 = points[i + 1];
    91→    91→    const p3 = points[i + 2] ?? p2;
    92→    92→    const cp1x = p1.x + ((p2.x - p0.x) * tension) / 6;
    93→    93→    const cp1y = p1.y + ((p2.y - p0.y) * tension) / 6;
    94→    94→    const cp2x = p2.x - ((p3.x - p1.x) * tension) / 6;
    95→    95→    const cp2y = p2.y - ((p3.y - p1.y) * tension) / 6;
    96→    96→    d += ` C ${cp1x} ${cp1y}, ${cp2x} ${cp2y}, ${p2.x} ${p2.y}`;
    97→    97→  }
    98→    98→  return d;
    99→    99→}

   107→   107→  const cx = size / 2;
   108→   108→  const cy = size / 2;

   110→   110→  const axes = RADAR_AXES;
   111→   111→  const n = axes.length;
   112→   112→
   113→   113→  // Pre-compute axis end points (used for both gridlines and labels)

   115→   115→    const angle = (Math.PI * 2 * i) / n - Math.PI / 2; // start at top
   116→   116→    return {
   117→   117→      x: cx + Math.cos(angle) * radius,
   118→   118→      y: cy + Math.sin(angle) * radius,
   119→   119→      angle,

   123→   123→  // Build polygon points string at a given set of 0-100 values
   124→   124→  const polygonPoints = (vals: number[]): string =>
   125→   125→    vals
   126→   126→      .map((v, i) => {
   127→   127→        const r = (Math.max(0, Math.min(100, v)) / 100) * radius;

   129→   129→        return `${cx + Math.cos(a) * r},${cy + Math.sin(a) * r}`;

   133→   133→  const candidateVals = axes.map((a) => values[a.key]);

   280→   280→
   281→   281→function TrajectorySparkline({
   282→   282→  evals,
   283→   283→}: {
   284→   284→  evals: AnswerEvaluation[];
   285→   285→}) {
   286→   286→  const width = 220;
   287→   287→  const height = 56;
   288→   288→  const padX = 8;
   289→   289→  const padY = 10;
   290→   290→
   291→   291→  const points = evals.map((e, i) => {
   292→   292→    const x =
   293→   293→      evals.length === 1
   294→   294→        ? width / 2
   295→   295→        : padX + (i / (evals.length - 1)) * (width - padX * 2);
   296→   296→    const y =
   297→   297→      height - padY - (Math.round(e.overall * 100) / 100) * (height - padY * 2);
   298→   298→    return { x, y };
   299→   299→  });
   300→   300→
   301→   301→  const path = smoothPath(points);
   302→   302→  const areaPath =
   303→   303→    points.length > 0
   304→   304→      ? `${path} L ${points[points.length - 1].x} ${height - padY} L ${points[0].x} ${height - padY} Z`
   305→   305→      : "";
   306→   306→
   307→   307→  const trend =
   308→   308→    evals.length < 2
   309→   309→      ? { dir: "flat", label: "Baseline", icon: Minus }
   310→   310→      : evals[evals.length - 1].overall > evals[0].overall + 0.02
   311→   311→      ? { dir: "up", label: "Improving", icon: TrendingUp }
   312→   312→      : evals[evals.length - 1].overall < evals[0].overall - 0.02
   313→   313→      ? { dir: "down", label: "Declining", icon: TrendingDown }
   314→   314→      : { dir: "flat", label: "Steady", icon: Minus };
   315→   315→
   316→   316→  const TrendIcon = trend.icon;
   317→   317→  const trendColor =
   318→   318→    trend.dir === "up"
   319→   319→      ? "text-success-foreground"
   320→   320→      : trend.dir === "down"
   321→   321→      ? "text-critical-foreground"
   322→   322→      : "text-muted-foreground";
   323→   323→
   324→   324→  return (
   325→   325→    <div className="flex flex-col gap-2">
   326→   326→      {evals.length < 2 ? (
   327→   327→        <div
   328→   328→          className="flex items-center justify-center rounded-md border border-dashed border-border/60 bg-muted/30 px-3 py-3 text-[11px] text-muted-foreground"
   329→   329→          style={{ width, height }}
   330→   330→        >
   331→   331→          <span className="inline-flex items-center gap-1.5">
   332→   332→            <span className="inline-block h-1.5 w-1.5 rounded-full bg-accent-blue" />
   333→   333→            Baseline established · trajectory appears after Q2
   334→   334→          </span>
   335→   335→        </div>
   336→   336→      ) : (
   337→   337→        <svg width={width} height={height} className="overflow-visible">
   338→   338→          <defs>
   339→   339→            <linearGradient id="hm-spark-area" x1="0" y1="0" x2="0" y2="1">
   340→   340→              <stop offset="0%" stopColor="var(--accent-blue)" stopOpacity={0.28} />
   341→   341→              <stop offset="100%" stopColor="var(--accent-blue)" stopOpacity={0} />
   342→   342→            </linearGradient>
   343→   343→          </defs>
   344→   344→          {/* Baseline at 50% */}
   345→   345→          <line
   346→   346→            x1={padX}
   347→   347→            x2={width - padX}
   348→   348→            y1={height / 2}
   349→   349→            y2={height / 2}
   350→   350→            stroke="var(--border)"
   351→   351→            strokeWidth={1}
   352→   352→            strokeDasharray="2 3"
   353→   353→            opacity={0.6}
   354→   354→          />
   355→   355→          {areaPath && (
   356→   356→            <motion.path
   357→   357→              initial={{ opacity: 0 }}
   358→   358→              animate={{ opacity: 1 }}
   359→   359→              transition={{ duration: 0.6, delay: 0.4 }}
   360→   360→              d={areaPath}
   361→   361→              fill="url(#hm-spark-area)"
   362→   362→            />
   363→   363→          )}
   364→   364→          {path && (
   365→   365→            <motion.path
   366→   366→              initial={{ pathLength: 0 }}
   367→   367→              animate={{ pathLength: 1 }}
   368→   368→              transition={{ duration: 0.9, delay: 0.25, ease: [0.22, 1, 0.36, 1] }}
   369→   369→              d={path}
   370→   370→              fill="none"
   371→   371→              stroke="var(--accent-blue)"
   372→   372→              strokeWidth={2}
   373→   373→              strokeLinecap="round"
   374→   374→              strokeLinejoin="round"
   375→   375→            />
   376→   376→          )}
   377→   377→          {points.map((p, i) => (
   378→   378→            <motion.circle
   379→   379→              key={i}
   380→   380→              initial={{ opacity: 0, r: 0 }}
   381→   381→              animate={{ opacity: 1, r: 2.8 }}
   382→   382→              transition={{ duration: 0.3, delay: 0.5 + i * 0.08 }}
   383→   383→              cx={p.x}
   384→   384→              cy={p.y}
   385→   385→              fill="var(--card)"
   386→   386→              stroke="var(--accent-blue)"
   387→   387→              strokeWidth={1.8}
   388→   388→            />
   389→   389→          ))}
   390→   390→        </svg>
   391→   391→      )}
   392→   392→      <div className="flex items-center justify-between text-[10px]">
   393→   393→        <span className="text-muted-foreground">
   394→   394→          Q1{evals.length > 1 ? ` → Q${evals.length}` : " · baseline"}
   395→   395→        </span>
   396→   396→        <span className={`inline-flex items-center gap-1 font-medium ${trendColor}`}>
   397→   397→          <TrendIcon className="h-3 w-3" />
   398→   398→          {trend.label}
   399→   399→        </span>
   400→   400→      </div>
   401→   401→    </div>
   402→   402→  );
   403→   403→}

   408→   408→
   409→   409→interface MatrixQuadrant {
   410→   410→  title: string;
   411→   411→  description: string;
   412→   412→  chips: string[];
   413→   413→  borderClass: string;
   414→   414→  chipClass: string;

   416→   416→  iconColor: string;
   417→   417→  icon: React.ComponentType<{ className?: string }>;
   418→   418→}
   419→   419→
   420→   420→function StrengthWeaknessMatrix({
   421→   421→  interview,
   422→   422→  gaps,
   423→   423→}: {
   424→   424→  interview: InterviewState;
   425→   425→  gaps: { competency: string; importance: string }[] | null;
   426→   426→}) {
   427→   427→  // Build competency chips per quadrant from the deterministic state.
   428→   428→  const states = interview.competencyStates;
   429→   429→  const evaluatedCompetencies = new Set(
   430→   430→    interview.evaluations.map((e) => e.competency)
   431→   431→  );
   432→   432→
   433→   433→  const strongPracticed: string[] = [];
   434→   434→  const strongUnverified: string[] = [];
   435→   435→  const weakImproving: string[] = [];
   436→   436→  const criticalGaps: string[] = [];
   437→   437→
   438→   438→  for (const s of states) {
   439→   439→    if (
   440→   440→      (s.interviewLevel === "strong" || s.current === "strong") &&
   441→   441→      evaluatedCompetencies.has(s.competency)
   442→   442→    ) {
   443→   443→      strongPracticed.push(s.competency);
   444→   444→    } else if (
   445→   445→      (s.resumeLevel === "strong" || s.resumeLevel === "moderate") &&
   446→   446→      s.interviewLevel === "unknown"
   447→   447→    ) {
   448→   448→      strongUnverified.push(s.competency);
   449→   449→    } else if (
   450→   450→      (s.interviewLevel === "weak" || s.current === "weak") &&
   451→   451→      s.status !== "gap"
   452→   452→    ) {
   453→   453→      weakImproving.push(s.competency);
   454→   454→    } else if (s.status === "gap" || s.current === "unknown") {
   455→   455→      criticalGaps.push(s.competency);
   456→   456→    }
   457→   457→  }
   458→   458→
   459→   459→  // Pull from gaps list to round out critical gaps
   460→   460→  if (gaps) {
   461→   461→    for (const g of gaps) {
   462→   462→      if (criticalGaps.length < 3 && !criticalGaps.includes(g.competency)) {
   463→   463→        if (g.importance === "critical" || g.importance === "high") {
   464→   464→          criticalGaps.push(g.competency);
   465→   465→        }
   466→   466→      }
   467→   467→    }
   468→   468→  }
   469→   469→
   470→   470→  // Limit each quadrant to 3 chips; use em-dash placeholder when empty so the
   471→   471→  // 2x2 grid stays balanced.

   561→   561→
   562→   562→function TimeAnalysis({ interview }: { interview: InterviewState }) {
   563→   563→  const minutes = interview.answers.map((a) => estimateMinutes(a.text));
   564→   564→  const maxMin = Math.max(1, ...minutes);
   565→   565→  const avg = minutes.length > 0 ? average(minutes) : 0;
   566→   566→
   567→   567→  const pace =
   568→   568→    avg >= 3
   569→   569→      ? { label: "Thoughtful", icon: Gauge, color: "text-success-foreground", bg: "bg-success/15" }
   570→   570→      : avg >= 1
   571→   571→      ? { label: "Quick", icon: Zap, color: "text-accent-blue-foreground", bg: "bg-accent-blue/15" }
   572→   572→      : { label: "Rushed", icon: Clock, color: "text-warning-foreground", bg: "bg-warning/15" };
   573→   573→
   574→   574→  const PaceIcon = pace.icon;
   575→   575→
   576→   576→  return (
   577→   577→    <div>
   578→   578→      <div className="space-y-2">
   579→   579→        {minutes.map((m, i) => {
   580→   580→          const widthPct = (m / maxMin) * 100;
   581→   581→          return (
   582→   582→            <motion.div
   583→   583→              key={i}
   584→   584→              initial={{ opacity: 0, x: -6 }}
   585→   585→              animate={{ opacity: 1, x: 0 }}
   586→   586→              transition={{ duration: 0.3, delay: i * 0.06 }}
   587→   587→              className="flex items-center gap-2"
   588→   588→            >
   589→   589→              <span className="text-[10px] font-medium text-muted-foreground w-6 shrink-0 tabular-nums">
   590→   590→                Q{i + 1}
   591→   591→              </span>
   592→   592→              <div className="flex-1 h-4 rounded bg-muted/60 overflow-hidden">
   593→   593→                <motion.div
   594→   594→                  initial={{ width: 0 }}
   595→   595→                  animate={{ width: `${widthPct}%` }}
   596→   596→                  transition={{ duration: 0.6, delay: 0.2 + i * 0.06, ease: [0.22, 1, 0.36, 1] }}
   597→   597→                  className="h-full rounded"
   598→   598→                  style={{
   599→   599→                    background:
   600→   600→                      m >= 3
   601→   601→                        ? "linear-gradient(90deg, var(--success), var(--accent-blue))"
   602→   602→                        : m >= 1
   603→   603→                        ? "linear-gradient(90deg, var(--accent-blue), var(--accent-blue))"
   604→   604→                        : "linear-gradient(90deg, var(--warning), var(--warning))",
   605→   605→                  }}
   606→   606→                />
   607→   607→              </div>
   608→   608→              <span className="text-[10px] text-muted-foreground w-12 text-right tabular-nums">
   609→   609→                {m}min
   610→   610→              </span>
   611→   611→            </motion.div>
   612→   612→          );
   613→   613→        })}
   614→   614→      </div>
   615→   615→      <div className="mt-3 flex items-center gap-2">
   616→   616→        <span className={`inline-flex items-center gap-1 rounded-full ${pace.bg} ${pace.color} px-2 py-0.5 text-[10px] font-semibold uppercase tracking-wider`}>
   617→   617→          <PaceIcon className="h-2.5 w-2.5" />
   618→   618→          {pace.label}
   619→   619→        </span>
   620→   620→        <span className="text-[10px] text-muted-foreground">
   621→   621→          avg {avg.toFixed(1)} min / answer
   622→   622→        </span>
   623→   623→      </div>
   624→   624→    </div>
   625→   625→  );
   626→   626→}
