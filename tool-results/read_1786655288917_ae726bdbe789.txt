    36→/** Map a 0..1 dimension score onto the smooth color scale:
    37→ * <0.30 critical · 0.30–0.50 warning · 0.50–0.70 accent-blue · >0.70 success. */
    38→function scoreToStatus(score: number): "matched" | "weak" | "gap" | "accent" {
    39→  if (score >= 0.7) return "matched";
    40→  if (score >= 0.5) return "accent";
    41→  if (score >= 0.3) return "weak";
    42→  return "gap";
    43→}

   501→function InsightsTabs() {
   502→  const { gaps, interview, match } = useHireMind();
   503→  const [tab, setTab] = React.useState("strengths");
   504→
   505→  // Derive strengths: competencies whose current level is strong or moderate,
   506→  // or that have interview evidence with high overall scores.
   507→  const strengths: { name: string; reason: string }[] = [];
   508→  const watchOuts: { name: string; reason: string }[] = [];
   509→
   510→  if (interview) {
   511→    for (const c of interview.competencyStates) {
   512→      if (c.current === "strong") {
   513→        strengths.push({
   514→          name: c.competency,
   515→          reason:
   516→            c.interviewLevel !== "unknown"
   517→              ? `Demonstrated in interview (resume: ${c.resumeLevel}).`
   518→              : `Strong resume signal — not yet verified in interview.`,
   519→        });
   520→      } else if (c.current === "moderate" && c.interviewLevel !== "unknown") {
   521→        strengths.push({
   522→          name: c.competency,
   523→          reason: `Solid interview evidence (resume: ${c.resumeLevel} → interview: ${c.interviewLevel}).`,
   524→        });
   525→      } else if (c.status === "gap" || c.current === "unknown" || c.current === "weak") {
   526→        watchOuts.push({
   527→          name: c.competency,
   528→          reason:
   529→            c.notes ||
   530→            (c.interviewLevel === "unknown"
   531→              ? "No interview evidence yet — unverified."
   532→              : "Low scores in this area."),
   533→        });
   534→      }
   535→    }
   536→  }
   537→
   538→  // Augment watch-outs from gaps list (if not already present)
   539→  if (gaps) {
   540→    for (const g of gaps) {
   541→      if (watchOuts.find((w) => w.name === g.competency)) continue;
   542→      if (g.importance === "critical" || g.importance === "high") {
   543→        watchOuts.push({
   544→          name: g.competency,
   545→          reason: g.reason,
   546→        });
   547→      }
   548→    }
   549→  }
   550→
   551→  // Augment strengths from match rows (matched competencies not in competencyStates)
   552→  if (match) {
   553→    for (const r of match.rows) {
   554→      if (r.status === "matched" && !strengths.find((s) => s.name === r.competency)) {
   555→        strengths.push({
   556→          name: r.competency,
   557→          reason: r.evidence
   558→            ? `Resume evidence: "${r.evidence.slice(0, 80)}${r.evidence.length > 80 ? "…" : ""}"`
   559→            : "Aligned with job requirements.",
   560→        });
   561→      }
   562→    }
   563→  }
   564→
   565→  // Cap each list at 6 items for layout stability
   566→  const topStrengths = strengths.slice(0, 6);
   567→  const topWatchOuts = watchOuts.slice(0, 6);
   568→
   569→  return (
   570→    <motion.div
   571→      initial={{ opacity: 0, y: 8 }}
   572→      animate={{ opacity: 1, y: 0 }}
   573→      transition={{ duration: 0.5, delay: 0.18 }}
   574→      className="hm-card hm-card-hover mt-4 p-4 sm:p-6 overflow-visible"
   575→    >
   576→      <div className="flex items-center gap-2.5 mb-4">
   577→        <span className="inline-flex h-9 w-9 items-center justify-center rounded-xl bg-accent-blue/15 text-accent-blue-foreground">
   578→          <LayoutGrid className="h-5 w-5" />
   579→        </span>
   580→        <div>
   581→          <h3 className="text-[13px] font-semibold">Insights</h3>
   582→          <div className="text-[11px] text-muted-foreground">A deeper breakdown of your readiness signals</div>
   583→        </div>
   584→      </div>
   585→
   586→      <Tabs value={tab} onValueChange={setTab}>
   587→        <TabsList className="mb-4">
   588→          <TabsTrigger value="strengths" className="gap-1.5">
   589→            <CheckCircle2 className="h-3 w-3" />
   590→            Strengths
   591→            <span className="ml-1 rounded bg-success/15 text-success-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   592→              {topStrengths.length}
   593→            </span>
   594→          </TabsTrigger>
   595→          <TabsTrigger value="watchouts" className="gap-1.5">
   596→            <AlertTriangle className="h-3 w-3" />
   597→            Watch-outs
   598→            <span className="ml-1 rounded bg-warning/15 text-warning-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   599→              {topWatchOuts.length}
   600→            </span>
   601→          </TabsTrigger>
   602→          <TabsTrigger value="coverage" className="gap-1.5">
   603→            <LayoutGrid className="h-3 w-3" />
   604→            Coverage
   605→          </TabsTrigger>
   606→        </TabsList>
   607→
   608→        {/* Strengths tab */}
   609→        <TabsContent value="strengths">
   610→          {topStrengths.length > 0 ? (
   611→            <ul className="grid sm:grid-cols-2 gap-2.5">
   612→              {topStrengths.map((s, i) => (
   613→                <motion.li
   614→                  key={s.name}
   615→                  initial={{ opacity: 0, y: 6 }}
   616→                  animate={{ opacity: 1, y: 0 }}
   617→                  transition={{ duration: 0.3, delay: i * 0.05 }}
   618→                  className="flex items-start gap-2.5 rounded-xl border border-success/25 bg-success/5 p-3"
   619→                >
   620→                  <span className="mt-0.5 inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-md bg-success/15 text-success-foreground">
   621→                    <CheckCircle2 className="h-3 w-3" />
   622→                  </span>
   623→                  <div className="min-w-0">
   624→                    <div className="text-[13px] font-semibold text-foreground leading-tight">{s.name}</div>
   625→                    <div className="mt-0.5 text-[11px] text-muted-foreground leading-relaxed">{s.reason}</div>
   626→                  </div>
   627→                </motion.li>
   628→              ))}
   629→            </ul>
   630→          ) : (
   631→            <p className="text-[13px] text-muted-foreground py-6 text-center">
   632→              No standout strengths detected yet. Run the interview to gather evidence.
   633→            </p>
   634→          )}
   635→        </TabsContent>
   636→
   637→        {/* Watch-outs tab */}
   638→        <TabsContent value="watchouts">
   639→          {topWatchOuts.length > 0 ? (
   640→            <ul className="grid sm:grid-cols-2 gap-2.5">
   641→              {topWatchOuts.map((w, i) => (
   642→                <motion.li
   643→                  key={w.name}
   644→                  initial={{ opacity: 0, y: 6 }}
   645→                  animate={{ opacity: 1, y: 0 }}
   646→                  transition={{ duration: 0.3, delay: i * 0.05 }}
   647→                  className="flex items-start gap-2.5 rounded-xl border border-warning/25 bg-warning/5 p-3"
   648→                >
   649→                  <span className="mt-0.5 inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-md bg-warning/15 text-warning-foreground">
   650→                    <AlertTriangle className="h-3 w-3" />
   651→                  </span>
   652→                  <div className="min-w-0">
   653→                    <div className="text-[13px] font-semibold text-foreground leading-tight">{w.name}</div>
   654→                    <div className="mt-0.5 text-[11px] text-muted-foreground leading-relaxed">{w.reason}</div>
   655→                  </div>
   656→                </motion.li>
   657→              ))}
   658→            </ul>
   659→          ) : (
   660→            <p className="text-[13px] text-muted-foreground py-6 text-center">
   661→              No watch-outs detected. Your profile aligns well with the role requirements.
   662→            </p>
   663→          )}
   664→        </TabsContent>
   665→
   666→        {/* Coverage tab — heatmap grid */}
   667→        <TabsContent value="coverage">
   668→          <CoverageHeatmap />
   669→        </TabsContent>

   671→    </motion.div>

   675→/** Coverage heatmap — grid of competency cells. Each cell shows the competency
   676→ *  name, level, and required badge. Background tint + dot color reflect the
   677→ *  candidate's current level. */
   678→function CoverageHeatmap() {
   679→  const { match, interview } = useHireMind();
   680→
   681→  // Merge match rows with interview competency states so each cell shows the
   682→  // latest known level (interview evidence takes precedence over resume level).
   683→  const interviewLevels = new Map<string, SkillLevel>();
   684→  if (interview) {
   685→    for (const c of interview.competencyStates) {
   686→      interviewLevels.set(c.competency, c.current);
   687→    }
   688→  }
   689→
   690→  if (!match || match.rows.length === 0) {
   691→    return (
   692→      <p className="text-[13px] text-muted-foreground py-6 text-center">
   693→        Run an analysis to see your competency coverage.
   694→      </p>
   695→    );
   696→  }
   697→
   698→  const rows = match.rows;
   699→
   700→  return (
   701→    <div>
   702→      <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   703→        <div className="text-[11px] text-muted-foreground">
   704→          {rows.length} competencies · color shows your level · ring shows required
   705→        </div>
   706→        <div className="flex items-center gap-3 text-[10px] text-muted-foreground">
   707→          {(["strong", "moderate", "weak", "unknown"] as SkillLevel[]).map((lvl) => (
   708→            <span key={lvl} className="inline-flex items-center gap-1">
   709→              <span className={`inline-block h-2 w-2 rounded-sm ${LEVEL_INTENSITY[lvl].dot}`} />
   710→              {LEVEL_INTENSITY[lvl].label}
   711→            </span>
   712→          ))}
   713→        </div>
   714→      </div>
   715→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-2">
   716→        {rows.map((r, i) => {
   717→          const level = interviewLevels.get(r.competency) ?? r.candidateLevel;
   718→          const intensity = LEVEL_INTENSITY[level];
   719→          return (
   720→            <motion.div
   721→              key={r.competency}
   722→              initial={{ opacity: 0, scale: 0.95 }}
   723→              animate={{ opacity: 1, scale: 1 }}
   724→              transition={{ duration: 0.3, delay: Math.min(i * 0.025, 0.4) }}
   725→              className={`relative rounded-lg border ${r.required ? "border-foreground/40" : "border-border/60"} ${intensity.bg} p-2.5 min-h-[68px] flex flex-col justify-between`}
   726→              title={`${r.competency} — ${intensity.label} (${r.importance} importance)`}
   727→            >
   728→              <div className="flex items-start justify-between gap-1">
   729→                <span className="text-[11px] font-semibold leading-tight line-clamp-2 text-foreground">
   730→                  {r.competency}
   731→                </span>
   732→                {r.required && (
   733→                  <span className="shrink-0 rounded bg-foreground/85 text-background px-1 py-0.5 text-[8px] font-bold uppercase tracking-wider leading-none">
   734→                    Req
   735→                  </span>
   736→                )}
   737→              </div>
   738→              <div className="flex items-center gap-1.5 mt-1.5">
   739→                <span className={`inline-block h-1.5 w-1.5 rounded-full ${intensity.dot}`} />
   740→                <span className={`text-[10px] font-medium ${intensity.text}`}>
   741→                  {intensity.label}
   742→                </span>
   743→              </div>
   744→            </motion.div>
   745→          );
   746→        })}
   747→      </div>
   748→    </div>
   749→  );
