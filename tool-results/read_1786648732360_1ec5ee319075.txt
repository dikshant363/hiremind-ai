    34→/** Map a 0..1 dimension score onto the smooth color scale:
    35→ * <0.30 critical · 0.30–0.50 warning · 0.50–0.70 accent-blue · >0.70 success. */
    36→function scoreToStatus(score: number): "matched" | "weak" | "gap" | "accent" {
    37→  if (score >= 0.7) return "matched";
    38→  if (score >= 0.5) return "accent";
    39→  if (score >= 0.3) return "weak";
    40→  return "gap";
    41→}

   491→function InsightsTabs() {
   492→  const { gaps, interview, match } = useHireMind();
   493→  const [tab, setTab] = React.useState("strengths");
   494→
   495→  // Derive strengths: competencies whose current level is strong or moderate,
   496→  // or that have interview evidence with high overall scores.
   497→  const strengths: { name: string; reason: string }[] = [];
   498→  const watchOuts: { name: string; reason: string }[] = [];
   499→
   500→  if (interview) {
   501→    for (const c of interview.competencyStates) {
   502→      if (c.current === "strong") {
   503→        strengths.push({
   504→          name: c.competency,
   505→          reason:
   506→            c.interviewLevel !== "unknown"
   507→              ? `Demonstrated in interview (resume: ${c.resumeLevel}).`
   508→              : `Strong resume signal — not yet verified in interview.`,
   509→        });
   510→      } else if (c.current === "moderate" && c.interviewLevel !== "unknown") {
   511→        strengths.push({
   512→          name: c.competency,
   513→          reason: `Solid interview evidence (resume: ${c.resumeLevel} → interview: ${c.interviewLevel}).`,
   514→        });
   515→      } else if (c.status === "gap" || c.current === "unknown" || c.current === "weak") {
   516→        watchOuts.push({
   517→          name: c.competency,
   518→          reason:
   519→            c.notes ||
   520→            (c.interviewLevel === "unknown"
   521→              ? "No interview evidence yet — unverified."
   522→              : "Low scores in this area."),
   523→        });
   524→      }
   525→    }
   526→  }
   527→
   528→  // Augment watch-outs from gaps list (if not already present)
   529→  if (gaps) {
   530→    for (const g of gaps) {
   531→      if (watchOuts.find((w) => w.name === g.competency)) continue;
   532→      if (g.importance === "critical" || g.importance === "high") {
   533→        watchOuts.push({
   534→          name: g.competency,
   535→          reason: g.reason,
   536→        });
   537→      }
   538→    }
   539→  }
   540→
   541→  // Augment strengths from match rows (matched competencies not in competencyStates)
   542→  if (match) {
   543→    for (const r of match.rows) {
   544→      if (r.status === "matched" && !strengths.find((s) => s.name === r.competency)) {
   545→        strengths.push({
   546→          name: r.competency,
   547→          reason: r.evidence
   548→            ? `Resume evidence: "${r.evidence.slice(0, 80)}${r.evidence.length > 80 ? "…" : ""}"`
   549→            : "Aligned with job requirements.",
   550→        });
   551→      }
   552→    }
   553→  }
   554→
   555→  // Cap each list at 6 items for layout stability
   556→  const topStrengths = strengths.slice(0, 6);
   557→  const topWatchOuts = watchOuts.slice(0, 6);
   558→
   559→  return (
   560→    <motion.div
   561→      initial={{ opacity: 0, y: 8 }}
   562→      animate={{ opacity: 1, y: 0 }}
   563→      transition={{ duration: 0.5, delay: 0.18 }}
   564→      className="hm-card hm-card-hover mt-4 p-4 sm:p-6 overflow-visible"
   565→    >
   566→      <div className="flex items-center gap-2.5 mb-4">
   567→        <span className="inline-flex h-9 w-9 items-center justify-center rounded-xl bg-accent-blue/15 text-accent-blue-foreground">
   568→          <LayoutGrid className="h-5 w-5" />
   569→        </span>
   570→        <div>
   571→          <h3 className="text-[13px] font-semibold">Insights</h3>
   572→          <div className="text-[11px] text-muted-foreground">A deeper breakdown of your readiness signals</div>
   573→        </div>
   574→      </div>
   575→
   576→      <Tabs value={tab} onValueChange={setTab}>
   577→        <TabsList className="mb-4">
   578→          <TabsTrigger value="strengths" className="gap-1.5">
   579→            <CheckCircle2 className="h-3 w-3" />
   580→            Strengths
   581→            <span className="ml-1 rounded bg-success/15 text-success-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   582→              {topStrengths.length}
   583→            </span>
   584→          </TabsTrigger>
   585→          <TabsTrigger value="watchouts" className="gap-1.5">
   586→            <AlertTriangle className="h-3 w-3" />
   587→            Watch-outs
   588→            <span className="ml-1 rounded bg-warning/15 text-warning-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   589→              {topWatchOuts.length}
   590→            </span>
   591→          </TabsTrigger>
   592→          <TabsTrigger value="coverage" className="gap-1.5">
   593→            <LayoutGrid className="h-3 w-3" />
   594→            Coverage
   595→          </TabsTrigger>
   596→        </TabsList>
   597→
   598→        {/* Strengths tab */}
   599→        <TabsContent value="strengths">
   600→          {topStrengths.length > 0 ? (
   601→            <ul className="grid sm:grid-cols-2 gap-2.5">
   602→              {topStrengths.map((s, i) => (
   603→                <motion.li
   604→                  key={s.name}
   605→                  initial={{ opacity: 0, y: 6 }}
   606→                  animate={{ opacity: 1, y: 0 }}
   607→                  transition={{ duration: 0.3, delay: i * 0.05 }}
   608→                  className="flex items-start gap-2.5 rounded-xl border border-success/25 bg-success/5 p-3"
   609→                >
   610→                  <span className="mt-0.5 inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-md bg-success/15 text-success-foreground">
   611→                    <CheckCircle2 className="h-3 w-3" />
   612→                  </span>
   613→                  <div className="min-w-0">
   614→                    <div className="text-[13px] font-semibold text-foreground leading-tight">{s.name}</div>
   615→                    <div className="mt-0.5 text-[11px] text-muted-foreground leading-relaxed">{s.reason}</div>
   616→                  </div>
   617→                </motion.li>
   618→              ))}
   619→            </ul>
   620→          ) : (
   621→            <p className="text-[13px] text-muted-foreground py-6 text-center">
   622→              No standout strengths detected yet. Run the interview to gather evidence.
   623→            </p>

   625→        </TabsContent>
   626→
   627→        {/* Watch-outs tab */}
   628→        <TabsContent value="watchouts">
   629→          {topWatchOuts.length > 0 ? (
   630→            <ul className="grid sm:grid-cols-2 gap-2.5">

   634→                  initial={{ opacity: 0, y: 6 }}
   635→                  animate={{ opacity: 1, y: 0 }}
   636→                  transition={{ duration: 0.3, delay: i * 0.05 }}

   665→/** Coverage heatmap — grid of competency cells. Each cell shows the competency
   666→ *  name, level, and required badge. Background tint + dot color reflect the
   667→ *  candidate's current level. */
   668→function CoverageHeatmap() {
   669→  const { match, interview } = useHireMind();
   670→
   671→  // Merge match rows with interview competency states so each cell shows the
   672→  // latest known level (interview evidence takes precedence over resume level).
   673→  const interviewLevels = new Map<string, SkillLevel>();
   674→  if (interview) {
   675→    for (const c of interview.competencyStates) {
   676→      interviewLevels.set(c.competency, c.current);
   677→    }
   678→  }
   679→
   680→  if (!match || match.rows.length === 0) {
   681→    return (
   682→      <p className="text-[13px] text-muted-foreground py-6 text-center">
   683→        Run an analysis to see your competency coverage.
   684→      </p>
   685→    );
   686→  }
   687→
   688→  const rows = match.rows;
   689→
   690→  return (
   691→    <div>
   692→      <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   693→        <div className="text-[11px] text-muted-foreground">
   694→          {rows.length} competencies · color shows your level · ring shows required
