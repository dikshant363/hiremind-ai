    31→/** Map a 0..1 dimension score onto the smooth color scale:
    32→ * <0.30 critical · 0.30–0.50 warning · 0.50–0.70 accent-blue · >0.70 success. */
    33→function scoreToStatus(score: number): "matched" | "weak" | "gap" | "accent" {
    34→  if (score >= 0.7) return "matched";
    35→  if (score >= 0.5) return "accent";
    36→  if (score >= 0.3) return "weak";
    37→  return "gap";
    38→}

   456→
   457→function InsightsTabs() {
   458→  const { gaps, interview, match } = useHireMind();
   459→  const [tab, setTab] = React.useState("strengths");
   460→
   461→  // Derive strengths: competencies whose current level is strong or moderate,
   462→  // or that have interview evidence with high overall scores.
   463→  const strengths: { name: string; reason: string }[] = [];
   464→  const watchOuts: { name: string; reason: string }[] = [];
   465→
   466→  if (interview) {
   467→    for (const c of interview.competencyStates) {
   468→      if (c.current === "strong") {
   469→        strengths.push({
   470→          name: c.competency,
   471→          reason:
   472→            c.interviewLevel !== "unknown"
   473→              ? `Demonstrated in interview (resume: ${c.resumeLevel}).`
   474→              : `Strong resume signal — not yet verified in interview.`,
   475→        });
   476→      } else if (c.current === "moderate" && c.interviewLevel !== "unknown") {
   477→        strengths.push({
   478→          name: c.competency,
   479→          reason: `Solid interview evidence (resume: ${c.resumeLevel} → interview: ${c.interviewLevel}).`,
   480→        });
   481→      } else if (c.status === "gap" || c.current === "unknown" || c.current === "weak") {
   482→        watchOuts.push({
   483→          name: c.competency,
   484→          reason:
   485→            c.notes ||
   486→            (c.interviewLevel === "unknown"
   487→              ? "No interview evidence yet — unverified."
   488→              : "Low scores in this area."),
   489→        });
   490→      }
   491→    }
   492→  }
   493→
   494→  // Augment watch-outs from gaps list (if not already present)
   495→  if (gaps) {
   496→    for (const g of gaps) {
   497→      if (watchOuts.find((w) => w.name === g.competency)) continue;
   498→      if (g.importance === "critical" || g.importance === "high") {
   499→        watchOuts.push({
   500→          name: g.competency,
   501→          reason: g.reason,
   502→        });
   503→      }
   504→    }
   505→  }
   506→
   507→  // Augment strengths from match rows (matched competencies not in competencyStates)
   508→  if (match) {
   509→    for (const r of match.rows) {
   510→      if (r.status === "matched" && !strengths.find((s) => s.name === r.competency)) {
   511→        strengths.push({
   512→          name: r.competency,
   513→          reason: r.evidence
   514→            ? `Resume evidence: "${r.evidence.slice(0, 80)}${r.evidence.length > 80 ? "…" : ""}"`
   515→            : "Aligned with job requirements.",
   516→        });
   517→      }
   518→    }
   519→  }
   520→
   521→  // Cap each list at 6 items for layout stability
   522→  const topStrengths = strengths.slice(0, 6);
   523→  const topWatchOuts = watchOuts.slice(0, 6);
   524→
   525→  return (
   526→    <motion.div
   527→      initial={{ opacity: 0, y: 8 }}
   528→      animate={{ opacity: 1, y: 0 }}
   529→      transition={{ duration: 0.5, delay: 0.18 }}
   530→      className="hm-card hm-card-hover mt-4 p-4 sm:p-6 overflow-visible"
   531→    >
   532→      <div className="flex items-center gap-2.5 mb-4">
   533→        <span className="inline-flex h-9 w-9 items-center justify-center rounded-xl bg-accent-blue/15 text-accent-blue-foreground">
   534→          <LayoutGrid className="h-5 w-5" />
   535→        </span>
   536→        <div>
   537→          <h3 className="text-[13px] font-semibold">Insights</h3>
   538→          <div className="text-[11px] text-muted-foreground">A deeper breakdown of your readiness signals</div>
   539→        </div>
   540→      </div>
   541→
   542→      <Tabs value={tab} onValueChange={setTab}>
   543→        <TabsList className="mb-4">
   544→          <TabsTrigger value="strengths" className="gap-1.5">
   545→            <CheckCircle2 className="h-3 w-3" />
   546→            Strengths
   547→            <span className="ml-1 rounded bg-success/15 text-success-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   548→              {topStrengths.length}
   549→            </span>
   550→          </TabsTrigger>
   551→          <TabsTrigger value="watchouts" className="gap-1.5">
   552→            <AlertTriangle className="h-3 w-3" />
   553→            Watch-outs
   554→            <span className="ml-1 rounded bg-warning/15 text-warning-foreground px-1 text-[9px] font-semibold leading-none py-0.5">
   555→              {topWatchOuts.length}
   556→            </span>
   557→          </TabsTrigger>
   558→          <TabsTrigger value="coverage" className="gap-1.5">
   559→            <LayoutGrid className="h-3 w-3" />
   560→            Coverage
   561→          </TabsTrigger>
   562→        </TabsList>
   563→
   564→        {/* Strengths tab */}
   565→        <TabsContent value="strengths">
   566→          {topStrengths.length > 0 ? (
   567→            <ul className="grid sm:grid-cols-2 gap-2.5">
   568→              {topStrengths.map((s, i) => (
   569→                <motion.li
   570→                  key={s.name}
   571→                  initial={{ opacity: 0, y: 6 }}
   572→                  animate={{ opacity: 1, y: 0 }}
   573→                  transition={{ duration: 0.3, delay: i * 0.05 }}
   574→                  className="flex items-start gap-2.5 rounded-xl border border-success/25 bg-success/5 p-3"
   575→                >
   576→                  <span className="mt-0.5 inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-md bg-success/15 text-success-foreground">
   577→                    <CheckCircle2 className="h-3 w-3" />
   578→                  </span>
   579→                  <div className="min-w-0">
   580→                    <div className="text-[13px] font-semibold text-foreground leading-tight">{s.name}</div>
   581→                    <div className="mt-0.5 text-[11px] text-muted-foreground leading-relaxed">{s.reason}</div>
   582→                  </div>
   583→                </motion.li>
   584→              ))}
   585→            </ul>
   586→          ) : (
   587→            <p className="text-[13px] text-muted-foreground py-6 text-center">
   588→              No standout strengths detected yet. Run the interview to gather evidence.
   589→            </p>
   590→          )}
   591→        </TabsContent>
   592→
   593→        {/* Watch-outs tab */}
   594→        <TabsContent value="watchouts">
   595→          {topWatchOuts.length > 0 ? (
   596→            <ul className="grid sm:grid-cols-2 gap-2.5">
   597→              {topWatchOuts.map((w, i) => (
   598→                <motion.li
   599→                  key={w.name}
   600→                  initial={{ opacity: 0, y: 6 }}
   601→                  animate={{ opacity: 1, y: 0 }}
   602→                  transition={{ duration: 0.3, delay: i * 0.05 }}
   603→                  className="flex items-start gap-2.5 rounded-xl border border-warning/25 bg-warning/5 p-3"
   604→                >
   605→                  <span className="mt-0.5 inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-md bg-warning/15 text-warning-foreground">
   606→                    <AlertTriangle className="h-3 w-3" />
   607→                  </span>
   608→                  <div className="min-w-0">
   609→                    <div className="text-[13px] font-semibold text-foreground leading-tight">{w.name}</div>
   610→                    <div className="mt-0.5 text-[11px] text-muted-foreground leading-relaxed">{w.reason}</div>
   611→                  </div>
   612→                </motion.li>
   613→              ))}
   614→            </ul>
   615→          ) : (
   616→            <p className="text-[13px] text-muted-foreground py-6 text-center">
   617→              No watch-outs detected. Your profile aligns well with the role requirements.
   618→            </p>
   619→          )}
   620→        </TabsContent>
   621→
   622→        {/* Coverage tab — heatmap grid */}
   623→        <TabsContent value="coverage">
   624→          <CoverageHeatmap />
   625→        </TabsContent>
   626→      </Tabs>
   627→    </motion.div>
   628→  );
   629→}

   631→/** Coverage heatmap — grid of competency cells. Each cell shows the competency
   632→ *  name, level, and required badge. Background tint + dot color reflect the
   633→ *  candidate's current level. */

   635→  const { match, interview } = useHireMind();
   636→
   637→  // Merge match rows with interview competency states so each cell shows the
   638→  // latest known level (interview evidence takes precedence over resume level).
   639→  const interviewLevels = new Map<string, SkillLevel>();
   640→  if (interview) {
   641→    for (const c of interview.competencyStates) {
   642→      interviewLevels.set(c.competency, c.current);
   643→    }
   644→  }
   645→
   646→  if (!match || match.rows.length === 0) {
   647→    return (
   648→      <p className="text-[13px] text-muted-foreground py-6 text-center">
   649→        Run an analysis to see your competency coverage.
   650→      </p>
   651→    );
   652→  }
   653→

   655→
   656→  return (
   657→    <div>
   658→      <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   659→        <div className="text-[11px] text-muted-foreground">
