   368→/* ---------------------------------------------------------------------------
   369→ * Progress trajectory — horizontal track with three nodes (Current / Target /
   370→ * Mastery) + an "You need to grow ~X points" delta callout.
   371→ * ------------------------------------------------------------------------- */
   372→function ProgressTrajectory({ gap }: { gap: SkillGap }) {
   373→  const currentPct = CANDIDATE_LEVEL_PCT[gap.candidateLevel] ?? 5;
   374→  const targetPct = IMPORTANCE_PCT[gap.importance] ?? 50;
   375→  const masteryPct = 95;
   376→  const delta = targetPct - currentPct;
   377→
   378→  // Sort nodes by position so chips read left-to-right along the bar.
   379→  const nodes = [
   380→    { label: "Current", value: currentPct, color: "var(--muted-foreground)" },
   381→    { label: "Target", value: targetPct, color: "var(--accent-blue)" },
   382→    { label: "Mastery", value: masteryPct, color: "var(--success)" },
   383→  ].sort((a, b) => a.value - b.value);
   384→
   385→  return (
   386→    <div className="space-y-3">
   387→      {/* Bar + node markers */}
   388→      <div className="relative h-5">
   389→        {/* Track */}
   390→        <div className="absolute top-1/2 left-0 right-0 h-1.5 -translate-y-1/2 rounded-full bg-muted overflow-hidden">
   391→          <motion.div
   392→            className="h-full rounded-full"
   393→            style={{
   394→              background:
   395→                "linear-gradient(90deg, var(--muted-foreground), var(--accent-blue))",
   396→            }}
   397→            initial={{ width: 0 }}
   398→            animate={{ width: `${targetPct}%` }}
   399→            transition={{ duration: 0.8, ease: [0.22, 1, 0.36, 1], delay: 0.2 }}
   400→          />
   401→        </div>
   402→        {/* Node dots */}
   403→        {nodes.map((n, i) => (
   404→          <motion.div
   405→            key={n.label}
   406→            className="absolute top-1/2 -translate-y-1/2 -translate-x-1/2"
   407→            style={{ left: `${n.value}%` }}
   408→            initial={{ opacity: 0, scale: 0.5 }}
   409→            animate={{ opacity: 1, scale: 1 }}
   410→            transition={{ duration: 0.35, delay: 0.35 + i * 0.1, ease: [0.22, 1, 0.36, 1] }}
   411→          >
   412→            <div
   413→              className="h-3.5 w-3.5 rounded-full border-2 border-background"
   414→              style={{ background: n.color, boxShadow: `0 0 0 1px ${n.color}` }}
   415→            />
   416→          </motion.div>
   417→        ))}
   418→      </div>
   419→
   420→      {/* Chip row — Current / Target / Mastery */}
   421→      <div className="flex items-center justify-between gap-2 flex-wrap">
   422→        {nodes.map((n) => (
   423→          <div key={n.label} className="flex items-center gap-1.5">
   424→            <span
   425→              className="h-2 w-2 rounded-full shrink-0"
   426→              style={{ background: n.color }}
   427→            />
   428→            <span className="text-[11px] text-muted-foreground">{n.label}</span>
   429→            <span className="text-[11px] font-semibold tabular-nums text-foreground">
   430→              {n.value}%
   431→            </span>
   432→          </div>
   433→        ))}
   434→      </div>
   435→
   436→      {/* Delta callout */}
   437→      <motion.div
   438→        className="hm-insight-callout flex items-start gap-2 px-3 py-2"
   439→        initial={{ opacity: 0, y: 6 }}
   440→        animate={{ opacity: 1, y: 0 }}
   441→        transition={{ duration: 0.4, delay: 0.55, ease: [0.22, 1, 0.36, 1] }}
   442→      >
   443→        <TrendingUp className="h-3.5 w-3.5 text-accent-blue-foreground shrink-0 mt-0.5" />
   444→        <span className="text-[12px] text-foreground/85 leading-relaxed">
   445→          {delta > 0 ? (
   446→            <>
   447→              You need to grow{" "}
   448→              <span className="font-semibold text-foreground">~{delta} points</span>{" "}
   449→              to reach the target bar for this role.
   450→            </>
   451→          ) : (
   452→            <>
   453→              You're already at or past the target bar — focus on{" "}
   454→              <span className="font-semibold text-foreground">depth & fluency</span>{" "}
   455→              under interview pressure.
   456→            </>
   457→          )}
   458→        </span>
   459→      </motion.div>
   460→    </div>
   461→  );
   462→}

   499→/* ---------------------------------------------------------------------------
   500→ * GapDeepDive — the modal component
   501→ * ------------------------------------------------------------------------- */
   502→export function GapDeepDive({
   503→  gap,
   504→  open,
   505→  onOpenChange,
   506→}: {
   507→  gap: SkillGap | null;
   508→  open: boolean;
   509→  onOpenChange: (open: boolean) => void;
   510→}) {
   511→  const { startInterview, loading, loadingStep } = useHireMind();
   512→
   513→  // Keep a "rendered gap" so the modal can play its exit animation even after
   514→  // the parent nulls out the gap prop on close. Without this, the dialog
   515→  // content would unmount abruptly mid-exit-transition.
   516→  const [renderedGap, setRenderedGap] = React.useState<SkillGap | null>(null);
   517→  React.useEffect(() => {
   518→    if (gap) setRenderedGap(gap);
   519→    // Intentionally do NOT clear when gap becomes null — the next non-null
   520→    // gap will overwrite renderedGap on the next open.
   521→  }, [gap]);
   522→
   523→  const displayGap = gap ?? renderedGap;
   524→
   525→  const handleStartInterview = React.useCallback(async () => {
   526→    onOpenChange(false);
   527→    // Defer the actual interview start a tick so the modal close animation
   528→    // can begin without the view swap racing it.
   529→    setTimeout(() => {
   530→      void startInterview();
   531→    }, 80);
   532→  }, [onOpenChange, startInterview]);
   533→
   534→  const handleAddToRoadmap = React.useCallback(() => {
   535→    toast("Already in your roadmap", {
   536→      description:
   537→        "Your roadmap is auto-generated from your skill gaps — no need to add it manually.",
   538→    });
   539→  }, []);
   540→
   541→  if (!displayGap) {
   542→    // Render an inert Dialog so Radix keeps the close transition wired even
   543→    // when there is nothing to show.
   544→    return (
   545→      <Dialog open={open} onOpenChange={onOpenChange}>
   546→        <DialogContent className="sr-only" aria-hidden />
   547→      </Dialog>
   548→    );
   549→  }
